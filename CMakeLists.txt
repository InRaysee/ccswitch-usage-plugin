cmake_minimum_required(VERSION 3.21)
project(CCSwitchUsageBalance LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0603 WIN32_LEAN_AND_MEAN)
set(SQLITE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/sqlite-amalgamation-3530400")
add_library(sqlite STATIC "${SQLITE_SOURCE_DIR}/sqlite3.c")
target_include_directories(sqlite PUBLIC "${SQLITE_SOURCE_DIR}")
add_library(quickjs STATIC third_party/quickjs/quickjs.c third_party/quickjs/dtoa.c
    third_party/quickjs/libregexp.c third_party/quickjs/libunicode.c)
target_include_directories(quickjs PUBLIC third_party/quickjs)
target_compile_definitions(quickjs PRIVATE _GNU_SOURCE QUICKJS_NG_BUILD)
add_library(balance_core STATIC src/BalanceCore.cpp src/BalanceDisplay.cpp src/BalanceConfig.cpp src/BalanceTooltip.cpp)
target_link_libraries(balance_core PUBLIC sqlite quickjs winhttp gdi32)
target_include_directories(balance_core PUBLIC src)
add_library(CCSwitchUsageBalance SHARED src/CCSwitchUsage.cpp src/ProviderBalance.cpp src/OptionsDialog.cpp src/Options.rc)
target_link_libraries(CCSwitchUsageBalance PRIVATE balance_core gdi32 shell32 comctl32)
set_target_properties(CCSwitchUsageBalance PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
if(MINGW)
    target_link_options(CCSwitchUsageBalance PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(balance_tests tests/balance_tests.cpp)
    target_link_libraries(balance_tests PRIVATE balance_core)
    add_executable(plugin_probe tests/plugin_probe.cpp)
    target_include_directories(plugin_probe PRIVATE src)
    target_link_libraries(plugin_probe PRIVATE gdi32)
    if(MINGW)
        target_link_options(balance_tests PRIVATE -static -static-libgcc -static-libstdc++)
        target_link_options(plugin_probe PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME balance_core COMMAND balance_tests)
    add_executable(settings_tests tests/settings_tests.cpp)
    target_link_libraries(settings_tests PRIVATE balance_core)
    if(MINGW)
        target_link_options(settings_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME balance_settings COMMAND settings_tests)
    add_executable(options_tests tests/options_tests.cpp src/Options.rc)
    target_link_libraries(options_tests PRIVATE balance_core comctl32 shell32)
    if(MINGW)
        target_link_options(options_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME balance_options COMMAND options_tests)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME balance_http COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py" $<TARGET_FILE:balance_tests>)
    endif()
endif()
